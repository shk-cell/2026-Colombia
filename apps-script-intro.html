  var W = 360, H = 380, CX = 180, CY = 200, R = 150;
  var COLORS = ['#1A73E8', '#D93025', '#188038', '#7B3FBF', '#00796B', '#C25E00'];
  var FONT_FAMILY = 'system-ui, "Segoe UI", Roboto, Arial, sans-serif';
  var FONT = 'bold 18px ' + FONT_FAMILY;
  var canvas = document.getElementById('wheel');
  var ctx = canvas.getContext('2d');
  var dpr = window.devicePixelRatio || 1;
  canvas.width = W * dpr;
  canvas.height = H * dpr;
  ctx.scale(dpr, dpr);

  var items = [];
  var rot = 0;
  var spinning = false;

  function readItems() {
    items = document.getElementById('items').value
      .split('\n')
      .map(function (t) { return t.trim(); })
      .filter(function (t) { return t; });
  }

  function draw() {
    ctx.clearRect(0, 0, W, H);

    if (items.length < 2) {
      ctx.beginPath();
      ctx.arc(CX, CY, R, 0, Math.PI * 2);
      ctx.fillStyle = '#D9DFEA';
      ctx.fill();
      ctx.fillStyle = '#5D6879';
      ctx.font = FONT;
      ctx.textAlign = 'center';
      ctx.textBaseline = 'middle';
      ctx.fillText('Enter at least 2 items', CX, CY);
      return;
    }

    var seg = (Math.PI * 2) / items.length;
    // Shrink the label font as slices get thinner (18px max, 8px min)
    var fontSize = Math.max(8, Math.min(18, Math.floor(R * 0.55 * seg)));
    var maxChars = fontSize >= 14 ? 10 : 14;
    items.forEach(function (text, i) {
      var a0 = rot + i * seg;
      var a1 = a0 + seg;
      var c = i % COLORS.length;
      if (i === items.length - 1 && c === 0 && i > 0) c = 2; // avoid matching the first slice's color

      ctx.beginPath();
      ctx.moveTo(CX, CY);
      ctx.arc(CX, CY, R, a0, a1);
      ctx.closePath();
      ctx.fillStyle = COLORS[c];
      ctx.fill();
      ctx.strokeStyle = '#fff';
      ctx.lineWidth = 2;
      ctx.stroke();

      ctx.save();
      ctx.translate(CX, CY);
      ctx.rotate(a0 + seg / 2);
      ctx.fillStyle = '#fff';
      ctx.font = 'bold ' + fontSize + 'px ' + FONT_FAMILY;
      ctx.textAlign = 'right';
      ctx.textBaseline = 'middle';
      ctx.fillText(text.length > maxChars ? text.slice(0, maxChars) + '…' : text, R - 16, 0);
      ctx.restore();
    });

    // Center circle
    ctx.beginPath();
    ctx.arc(CX, CY, 18, 0, Math.PI * 2);
    ctx.fillStyle = '#fff';
    ctx.fill();

    // Pointer at the top
    ctx.beginPath();
    ctx.moveTo(CX - 14, 22);
    ctx.lineTo(CX + 14, 22);
    ctx.lineTo(CX, 62);
    ctx.closePath();
    ctx.fillStyle = '#172033';
    ctx.fill();
  }

  function spin() {
    if (spinning) return;
    readItems();
    if (items.length < 2) { draw(); return; }

    spinning = true;
    document.getElementById('spin').disabled = true;
    document.getElementById('result').textContent = '';

    var start = rot;
    var total = Math.PI * 2 * (5 + Math.random() * 3); // 5–8 turns plus a random angle
    var duration = 4000;
    var t0 = null;

    function step(ts) {
      if (t0 === null) t0 = ts;
      var p = Math.min((ts - t0) / duration, 1);
      var eased = 1 - Math.pow(1 - p, 3); // slow down gradually
      rot = start + total * eased;
      draw();
      if (p < 1) {
        requestAnimationFrame(step);
      } else {
        finish();
      }
    }
    requestAnimationFrame(step);
  }

  function finish() {
    var seg = (Math.PI * 2) / items.length;
    var pointer = Math.PI * 1.5; // the pointer sits at the top (12 o'clock)
    var angle = (((pointer - rot) % (Math.PI * 2)) + Math.PI * 2) % (Math.PI * 2);
    var index = Math.floor(angle / seg);
    document.getElementById('result').textContent = '🎉 ' + items[index];
    document.getElementById('spin').disabled = false;
    spinning = false;
  }

  document.getElementById('items').addEventListener('input', function () {
    if (spinning) return;
    readItems();
    document.getElementById('result').textContent = '';
    draw();
  });

  readItems();
  draw();

// Class Board (Google Sheets + Apps Script practice)
//
// - Writing a post: anyone
// - Reading posts / commenting / deleting: teacher only (password required)
// ★ Be sure to change this to a password only you know.
const TEACHER_PASSWORD = '1234';

// Spreadsheet ID (optional)
// If you opened this from the spreadsheet via [Extensions → Apps Script], leave it empty.
// If you created the project directly at script.google.com, paste the long string
// between /d/ and /edit in the spreadsheet's URL here.
const SPREADSHEET_ID = '';

// Name of the sheet where posts are stored (created automatically if missing)
const SHEET_NAME = 'Board';

// Time zone
const TIME_ZONE = 'America/Bogota';

// ---------------------------------------------------------------
// Open the web app
// ---------------------------------------------------------------
function doGet() {
  return HtmlService.createHtmlOutputFromFile('index')
    .setTitle('Class Board')
    .addMetaTag('viewport', 'width=device-width, initial-scale=1');
}

// ---------------------------------------------------------------
// What anyone can do: leave a post
// ---------------------------------------------------------------
function addPost(name, title, content) {
  name = clean_(name, 20, 'Name');
  title = clean_(title, 50, 'Title');
  content = clean_(content, 1000, 'Content');

  const lock = LockService.getScriptLock();
  lock.waitLock(10000); // handle posts one at a time so simultaneous writes don't collide
  try {
    const sheet = getSheet_();
    const row = sheet.getLastRow() + 1;
    const id = row === 2 ? 1 : Number(sheet.getRange(row - 1, 1).getValue()) + 1;
    const time = Utilities.formatDate(new Date(), TIME_ZONE, 'yyyy-MM-dd HH:mm');

    const range = sheet.getRange(row, 1, 1, 6);
    range.setNumberFormat('@'); // store as plain text (text starting with = or + won't become a formula)
    range.setValues([[String(id), time, name, title, content, '']]);
  } finally {
    lock.releaseLock();
  }
  return true;
}

// ---------------------------------------------------------------
// What only the teacher can do
// ---------------------------------------------------------------

// Read the list of posts (newest first)
function getPosts(password) {
  checkTeacher_(password);
  const sheet = getSheet_();
  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return [];

  const values = sheet.getRange(2, 1, lastRow - 1, 6).getValues();
  return values
    .map(function (r) {
      return {
        id: String(r[0]),
        time: String(r[1]),
        name: String(r[2]),
        title: String(r[3]),
        content: String(r[4]),
        comment: String(r[5])
      };
    })
    .reverse();
}

// Add a comment (multiple comments stack up underneath)
function addComment(password, id, text) {
  checkTeacher_(password);
  text = clean_(text, 300, 'Comment');

  const lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    const sheet = getSheet_();
    const row = findRow_(sheet, id);
    if (row === -1) throw new Error('Post not found. It may have already been deleted.');

    const cell = sheet.getRange(row, 6);
    const stamp = Utilities.formatDate(new Date(), TIME_ZONE, 'MM-dd HH:mm');
    const line = '[' + stamp + '] ' + text;
    const old = String(cell.getValue());
    cell.setNumberFormat('@');
    cell.setValue(old ? old + '\n' + line : line);
  } finally {
    lock.releaseLock();
  }
  return true;
}

// Delete a post
function deletePost(password, id) {
  checkTeacher_(password);

  const lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    const sheet = getSheet_();
    const row = findRow_(sheet, id);
    if (row === -1) throw new Error('Post not found. It may have already been deleted.');
    sheet.deleteRow(row);
  } finally {
    lock.releaseLock();
  }
  return true;
}

// ---------------------------------------------------------------
// Helper functions (a name ending in _ can't be called from the page)
// ---------------------------------------------------------------

// Check the password — throws an error if wrong, so the code below it never runs
function checkTeacher_(password) {
  if (password !== TEACHER_PASSWORD) {
    throw new Error('Incorrect password.');
  }
}

// Get the board sheet (creates it with a header row if missing)
function getSheet_() {
  const ss = SPREADSHEET_ID
    ? SpreadsheetApp.openById(SPREADSHEET_ID)
    : SpreadsheetApp.getActiveSpreadsheet();
  if (!ss) {
    throw new Error('Spreadsheet not found. Check SPREADSHEET_ID at the top of Code.gs.');
  }
  let sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) {
    sheet = ss.insertSheet(SHEET_NAME);
    sheet.appendRow(['No.', 'Date', 'Name', 'Title', 'Content', 'Comments']);
    sheet.setFrozenRows(1);
  }
  return sheet;
}

// Find the row number of a post by its id (-1 if not found)
function findRow_(sheet, id) {
  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return -1;
  const ids = sheet.getRange(2, 1, lastRow - 1, 1).getValues();
  for (let i = 0; i < ids.length; i++) {
    if (String(ids[i][0]) === String(id)) return i + 2;
  }
  return -1;
}

// Tidy up input: trim spaces, check it's not empty or too long
function clean_(value, max, label) {
  const text = String(value || '').trim();
  if (!text) throw new Error('Please fill in the ' + label.toLowerCase() + '.');
  if (text.length > max) throw new Error(label + ' can be up to ' + max + ' characters.');
  return text;
}

function getSpreadsheet_() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  if (ss) return ss;
  var props = PropertiesService.getScriptProperties();
  var id = props.getProperty('SPREADSHEET_ID');

function getSheet_() {
  var ss = getSpreadsheet_();
  var sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) {
    sheet = ss.insertSheet(SHEET_NAME);
    sheet.appendRow(['Name', 'Time (s)', 'Date']);

    sheet.getRange('A:A').setNumberFormat('@'); // keep names as plain text
  }

// Saves one run and returns the updated ranking plus this player's position.
function saveTime(name, seconds) {
  // Strip leading = + - @ so a name can never become a formula.
  var safeName = String(name || '').replace(/^[\s=+\-@]+/, '').trim().slice(0, 20) || 'Anonymous';
  var safeTime = Math.max(0, Math.round((Number(seconds) || 0) * 10) / 10);

  var lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    getSheet_().appendRow([safeName, safeTime, new Date()]);
    SpreadsheetApp.flush();
  } finally {
    lock.releaseLock();
  }
  return buildRanking_(safeName);
}

function getRanking() {
  return buildRanking_(null);
}

// Best time per player, longest survival first; ties go to whoever got there first.
function buildRanking_(playerName) {
  var sheet = getSheet_();
  var lastRow = sheet.getLastRow();
  var best = {};
  if (lastRow >= 2) {
    var rows = sheet.getRange(2, 1, lastRow - 1, 3).getValues();
    rows.forEach(function (r) {
      var name = String(r[0]).trim();
      var time = Number(r[1]);
      var at = r[2] instanceof Date ? r[2].getTime() : 0;
      if (!name || isNaN(time)) return;
      var cur = best[name];
      if (!cur || time > cur.time || (time === cur.time && at < cur.at)) {
        best[name] = { name: name, time: time, at: at };
      }
    });
  }

  var list = Object.keys(best).map(function (k) { return best[k]; });
  list.sort(function (a, b) { return b.time - a.time || a.at - b.at; });

  var myRank = 0, myBest = 0;
  if (playerName) {
    for (var i = 0; i < list.length; i++) {
      if (list[i].name === playerName) { myRank = i + 1; myBest = list[i].time; break; }
    }
  }

  // Only plain strings/numbers: google.script.run returns null if a Date is included.
  return {
    top: list.slice(0, TOP_N).map(function (e) { return { name: e.name, time: e.time }; }),
    total: list.length,
    me: playerName || '',
    myRank: myRank,
    myBest: myBest
  };
}

  canvas.addEventListener('mousedown', movePlayerTo);
  canvas.addEventListener('mousemove', function (e) { if (running) movePlayerTo(e); });
  canvas.addEventListener('touchstart', function (e) { movePlayerTo(e); }, { passive: true });
  canvas.addEventListener('touchmove', function (e) { e.preventDefault(); movePlayerTo(e); }, { passive: false });

  function spawnMissile() {
    var edge = Math.floor(Math.random() * 4);
    var x, y;
    if (edge === 0) { x = Math.random() * W; y = -20; }
    else if (edge === 1) { x = W + 20; y = Math.random() * H; }
    else if (edge === 2) { x = Math.random() * W; y = H + 20; }
    else { x = -20; y = Math.random() * H; }
    var angle = Math.atan2(player.y - y, player.x - x) + (Math.random() - 0.5) * 0.7;
    missiles.push({
      x: x, y: y, r: MISSILE_R,
      vx: Math.cos(angle) * missileSpeed,
      vy: Math.sin(angle) * missileSpeed,
      trail: []
    });
  }

  function updateHud() {
    document.getElementById('timeLabel').textContent = (elapsed / 1000).toFixed(1);
  }

  function showScreen(id) {
    ['startScreen', 'gameScreen', 'overScreen'].forEach(function (s) {
      document.getElementById(s).classList.toggle('show', s === id);
    });
  }

  /* ---------- Ranking ---------- */

  function renderBoard(listId, totalId, data) {
    var top = (data && data.top) || [];
    var medals = ['🥇', '🥈', '🥉'];
    var html = '';
    top.forEach(function (row, i) {
      var me = data.me && row.name === data.me;
      html += '<li' + (me ? ' class="me"' : '') + '>' +
        '<span class="rk">' + (medals[i] || (i + 1)) + '</span>' +
        '<span class="nm">' + escapeHtml(row.name) + (me ? ' (you)' : '') + '</span>' +
        '<span class="tm">' + Number(row.time).toFixed(1) + 's</span></li>';
    });
    if (!top.length) html = '<li class="msg">No records yet. Be the first!</li>';
    document.getElementById(listId).innerHTML = html;
    document.getElementById(totalId).textContent = data && data.total ? data.total + ' players' : '';
  }

  function boardMessage(listId, msg, isError) {
    document.getElementById(listId).innerHTML =
      '<li class="msg' + (isError ? ' err' : '') + '">' + escapeHtml(msg) + '</li>';
  }

  function loadStartRanking() {
    boardMessage('startBoard', 'Loading ranking…');
    google.script.run
      .withSuccessHandler(function (data) { renderBoard('startBoard', 'startTotal', data); })
      .withFailureHandler(function (err) { boardMessage('startBoard', 'Could not load ranking: ' + err.message, true); })
      .getRanking();
  }

  function escapeHtml(s) {
    return String(s).replace(/[&<>"']/g, function (c) {
      return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[c];
    });
  }

  /* ---------- Game flow ---------- */

  function startGame() {
    playerName = nameInput.value.trim().slice(0, 10) || 'Anonymous';
    try { localStorage.setItem('md-name', playerName); } catch (e) {}
    missiles = []; elapsed = 0; spawnTimer = START_INTERVAL;
    missileSpeed = START_SPEED; homingRate = START_HOMING;
    player.x = W / 2; player.y = H - 60;
    document.getElementById('bestLabel').textContent = personalBest ? personalBest.toFixed(1) : '–';
    updateHud();
    showScreen('gameScreen');
    fitCanvas();
    draw();
    running = true; lastTime = null;
    requestAnimationFrame(loop);
  }

  function endGame() {
    running = false;
    var seconds = Math.round(elapsed / 100) / 10;
    var isBest = seconds > personalBest;
    if (isBest) personalBest = seconds;

    document.getElementById('finalTime').textContent = seconds.toFixed(1);
    document.getElementById('bestBadge').classList.toggle('show', isBest && seconds > 0);
    document.getElementById('rankLine').textContent = 'Saving your record…';
    document.getElementById('overTotal').textContent = '';
    boardMessage('overBoard', 'Loading ranking…');
    showScreen('overScreen');

    // Save first, then show the ranking that the save returns — so your run is always included.
    google.script.run
      .withSuccessHandler(function (data) {
        renderBoard('overBoard', 'overTotal', data);
        if (data && data.myBest) personalBest = Math.max(personalBest, data.myBest);
        document.getElementById('rankLine').innerHTML = data && data.myRank
          ? escapeHtml(data.me) + ' is ranked <strong>#' + data.myRank + '</strong> of ' + data.total +
            ' · best ' + Number(data.myBest).toFixed(1) + 's'
          : '';
      })
      .withFailureHandler(function (err) {
        document.getElementById('rankLine').textContent = '';
        boardMessage('overBoard', 'Could not save your record: ' + err.message, true);
      })
      .saveTime(playerName, seconds);
  }

  function step(dt) {
    elapsed += dt;
    var elapsedSec = elapsed / 1000;
    var dtSec = dt / 1000;

    spawnInterval = Math.max(MIN_INTERVAL, START_INTERVAL - elapsedSec * INTERVAL_DECAY_PER_SEC);
    missileSpeed = Math.min(MAX_SPEED, START_SPEED + elapsedSec * SPEED_GROWTH_PER_SEC);
    homingRate = Math.min(MAX_HOMING, START_HOMING + elapsedSec * HOMING_GROWTH_PER_SEC);

    spawnTimer -= dt;
    if (spawnTimer <= 0) {
      var count = Math.min(MAX_SPAWN_PER_WAVE, 1 + Math.floor(elapsedSec / 10));
      for (var c = 0; c < count; c++) spawnMissile();
      spawnTimer = spawnInterval;
    }

    for (var i = missiles.length - 1; i >= 0; i--) {
      var m = missiles[i];
      var desired = Math.atan2(player.y - m.y, player.x - m.x);
      var cur = Math.atan2(m.vy, m.vx);
      var diff = Math.atan2(Math.sin(desired - cur), Math.cos(desired - cur));
      var maxTurn = homingRate * dtSec;
      var turn = Math.max(-maxTurn, Math.min(maxTurn, diff));
      var newAngle = cur + turn;
      var spd = Math.hypot(m.vx, m.vy);
      m.vx = Math.cos(newAngle) * spd;
      m.vy = Math.sin(newAngle) * spd;
      m.trail.push({ x: m.x, y: m.y });
      if (m.trail.length > 8) m.trail.shift();
      m.x += m.vx * dtSec;
      m.y += m.vy * dtSec;

      if (m.x < -40 || m.x > W + 40 || m.y < -40 || m.y > H + 40) {
        missiles.splice(i, 1);
        continue;
      }
      var dx = m.x - player.x, dy = m.y - player.y;
      if (Math.hypot(dx, dy) < m.r + player.r) {
        draw();
        endGame();
        return;
      }
    }

    updateHud();
  }

  /* ---------- Drawing ---------- */

  function drawGrid() {
    ctx.strokeStyle = '#EDF1F7';
    ctx.lineWidth = 1;
    ctx.beginPath();
    for (var x = 20; x < W; x += 20) { ctx.moveTo(x + 0.5, 0); ctx.lineTo(x + 0.5, H); }
    for (var y = 20; y < H; y += 20) { ctx.moveTo(0, y + 0.5); ctx.lineTo(W, y + 0.5); }
    ctx.stroke();
  }

  function drawPlayer() {
    ctx.beginPath();
    ctx.arc(player.x, player.y, player.r + 7, 0, Math.PI * 2);
    ctx.fillStyle = 'rgba(26, 95, 208, 0.12)';
    ctx.fill();
    ctx.beginPath();
    ctx.arc(player.x, player.y, player.r, 0, Math.PI * 2);
    ctx.fillStyle = '#1A5FD0';
    ctx.fill();
    ctx.lineWidth = 3;
    ctx.strokeStyle = '#FFFFFF';
    ctx.stroke();
  }

  function drawMissile(m) {
    for (var t = 0; t < m.trail.length; t++) {
      var p = m.trail[t];
      ctx.beginPath();
      ctx.arc(p.x, p.y, 1 + t * 0.3, 0, Math.PI * 2);
      ctx.fillStyle = 'rgba(217, 48, 37, ' + (0.04 + t * 0.03) + ')';
      ctx.fill();
    }
    var angle = Math.atan2(m.vy, m.vx);
    ctx.save();
    ctx.translate(m.x, m.y);
    ctx.rotate(angle);
    ctx.beginPath();
    ctx.moveTo(m.r * 1.7, 0);
    ctx.lineTo(-m.r, m.r * 0.8);
    ctx.lineTo(-m.r * 0.4, 0);
    ctx.lineTo(-m.r, -m.r * 0.8);
    ctx.closePath();
    ctx.fillStyle = '#D93025';
    ctx.fill();
    ctx.restore();
  }

  function draw() {
    ctx.clearRect(0, 0, W, H);
    drawGrid();
    missiles.forEach(drawMissile);
    drawPlayer();
  }

  function loop(ts) {
    if (!running) return;
    if (lastTime === null) lastTime = ts;
    var dt = Math.min(33, ts - lastTime);
    lastTime = ts;
    step(dt);
    if (running) {
      draw();
      requestAnimationFrame(loop);
    }
  }

  document.getElementById('startBtn').addEventListener('click', startGame);
  nameInput.addEventListener('keydown', function (e) { if (e.key === 'Enter') startGame(); });
  document.getElementById('retryBtn').addEventListener('click', startGame);
  document.getElementById('menuBtn').addEventListener('click', function () {
    showScreen('startScreen');
    loadStartRanking();
  });

  loadStartRanking();
})();
